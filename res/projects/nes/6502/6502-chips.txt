// ALU.hdl
CHIP ALU

IN
phi1, phi2
sb_bus[8], db_bus[8], adl_bus[8]
ndb_add, db_add, zero_add, sb_add, adl_add
sums, ands, eors, ors, srs
sb_ac
acin

OUT
ac_reg[8], add_reg[8]
acr, avr

PARTS
// =====================================================================================================================
// AI LATCH
// =====================================================================================================================
Mux<8>(a=ai_reg, b=sb_bus, sel=sb_add, out=ai_mux1)
Mux<8>(a=ai_mux1, b=8'd0, sel=zero_add, out=ai_next)

Or<1>(a=sb_add, b=zero_add, out=ai_load)
Register<8>(in=ai_next, load=ai_load, out=ai_reg)

// =====================================================================================================================
// BI LATCH
// Note: ndb_add loads the inverted DB bus to perform SBC (Subtraction) via 1's complement
// =====================================================================================================================
Not<8>(in=db_bus, out=ndb_bus)

Mux<8>(a=bi_reg, b=adl_bus, sel=adl_add, out=bi_mux1)
Mux<8>(a=bi_mux1, b=db_bus, sel=db_add, out=bi_mux2)
Mux<8>(a=bi_mux2, b=ndb_bus, sel=ndb_add, out=bi_next)

Or<1>(a=db_add, b=adl_add, out=bi_load_1)
Or<1>(a=bi_load_1, b=ndb_add, out=bi_load)
Register<8>(in=bi_next, load=bi_load, out=bi_reg)

// =====================================================================================================================
// ACCUMULATOR (AC)
// Anatomically embedded alongside the ALU; directly catches SB Bus
// =====================================================================================================================
Register<8>(in=sb_bus, load=sb_ac, out=ac_reg)

// =====================================================================================================================
// OPERATIONS
// =====================================================================================================================
// 9-Bit Addition for Carry Propagation (AI + BI + CarryIn)
Add<9>(a={1'b0, ai_reg}, b={1'b0, bi_reg}, out=sum_ab)
Add<9>(a=sum_ab, b={8'd0, acin}, out=sum_full)
sum_res = sum_full[0:7]
c_out = sum_full[8]

// Logical Operations
And<8>(a=ai_reg, b=bi_reg, out=and_res)
Or<8>(a=ai_reg, b=bi_reg, out=or_res)
Xor<8>(a=ai_reg, b=bi_reg, out=xor_res)

// Shift Right (LSR/ROR). The top bit inherits the carry-in flag.
sr_res = {acin, ai_reg[1:7]}

// =====================================================================================================================
// OUTPUT MULTIPLEXING & LATCH (ADD_REG)
// =====================================================================================================================
Mux<8>(a=8'd0, b=sum_res, sel=sums, out=res_1)
Mux<8>(a=res_1, b=and_res, sel=ands, out=res_2)
Mux<8>(a=res_2, b=xor_res, sel=eors, out=res_3)
Mux<8>(a=res_3, b=or_res, sel=ors, out=res_4)
Mux<8>(a=res_4, b=sr_res, sel=srs, out=add_next)

Or<1>(a=sums, b=ands, out=add_ld_1)
Or<1>(a=eors, b=ors, out=add_ld_2)
Or<1>(a=add_ld_1, b=add_ld_2, out=add_ld_3)
Or<1>(a=add_ld_3, b=srs, out=add_load)

Register<8>(in=add_next, load=add_load, out=add_reg)

// =====================================================================================================================
// FLAGS CALCULATION (ACR / AVR)
// =====================================================================================================================
// Overflow (V) = ~(A[7] ^ B[7]) & (A[7] ^ Sum[7])
// Note: For SBC, BI[7] is safely already inverted by ndb_add.
Xor<1>(a=ai_reg[7], b=bi_reg[7], out=xor_ab7)
Not<1>(in=xor_ab7, out=nxor_ab7)
Xor<1>(a=ai_reg[7], b=sum_res[7], out=xor_as7)
And<1>(a=nxor_ab7, b=xor_as7, out=avr_calc)

// Gate AVR to only output during Addition/Subtraction
Mux<1>(a=1'b0, b=avr_calc, sel=sums, out=avr)

// Carry (C) outputs arithmetic carry-out on sums, or the shifted-out bit 0 on SRS
Mux<1>(a=1'b0, b=ai_reg[0], sel=srs, out=acr_srs)
Mux<1>(a=acr_srs, b=c_out, sel=sums, out=acr)
---

// CPU.hdl
CHIP CPU  // MOS 6502 Core

@const(cpu_constants)

IN
data_in[8]
_nmi  // Non-maskable interrupt signal
_irq  // Maskable interrupt signal
_res  // Reset signal
phi0  // Reference clock signal
rdy   // Processor ready (1 = ready)
so    // Forced setting of the overflow flag (V)

OUT
addr[16]
data_out[8]
phi1  // First half-cycle, processor in writing mode (= ¬phi0)
phi2  // Second half-cycle, processor in read mode   (=  phi0)
rw    // Data bus direction (1 = reading, 0 = writing)
sync  // The processor is on cycle T1 (opcode fetch)

PARTS

// =====================================================================================================================
// CLOCK GENERATOR
// =====================================================================================================================
Not<1>(in=phi0, out=phi1)
phi2 = phi0


// =====================================================================================================================
// PREDECODE LOGIC: Determines instruction class from PD latch
// =====================================================================================================================

// Predecode (PD) latch grabs data on PHI2
Register<8>(in=data_in, load=phi2, out=pd)

// 2-cycle instructions
Eq<4>(a={pd[4], pd[3], pd[2], pd[0]}, b=MASK_2C_A, out=is_2c_a)
Eq<5>(a={pd[7], pd[4], pd[3], pd[2], pd[0]}, b=MASK_2C_B, out=is_2c_b)
// Single-byte instructions
Eq<3>(a={pd[3], pd[2], pd[0]}, b=MASK_IMPLIED, out=implied)
// Push/Pull
Eq<3>(a={pd[7], pd[4], pd[1]}, b=MASK_PUSHPULL, out=is_pushpull)
Not<1>(in=is_pushpull, out=not_pushpull)

// TWOCYCLE flag
And<1>(a=implied, b=not_pushpull, out=implied_valid)
Or<1>(a=is_2c_a, b=is_2c_b, out=is_2c)
Or<1>(a=is_2c, b=implied_valid, out=twocycle)
Not<1>(in=twocycle, out=n_twocycle)


// =====================================================================================================================
// INTERRUPT AND BRK SEQUENCE LOGIC
// =====================================================================================================================
Not<1>(in=_irq, out=irq)
Not<1>(in=_nmi, out=nmi)
Not<1>(in=_res, out=res)

// NMI edge detect: Detects falling edge of _nmi and latches it until interrupt sequence begins
Dff<1>(in=nmi, out=nmi_sync)
Not<1>(in=nmi_sync, out=n_nmi_sync)
And<1>(a=nmi, b=n_nmi_sync, out=nmi_edge)

Or<1>(a=nmi_edge, b=nmi_pending, out=nmi_set)
Not<1>(in=b_out, out=n_b_out) // Clear pending NMI when B_OUT (interrupt start) fires
And<1>(a=nmi_set, b=n_b_out, out=next_nmi_pending)
Dff<1>(in=next_nmi_pending, out=nmi_pending)

// IRQ and reset check
Not<1>(in=i_flag, out=n_i_flag)
And<1>(a=irq, b=n_i_flag, out=irq_pending)

// DORES: Synchronized reset flag
Register<1>(in=res, load=rdy, out=dores)

// Interrupt arrived, so force execution of BRK by injecting 0x00 into the IR
Or<1>(a=nmi_pending, b=irq_pending, out=any_int)
Or<1>(a=any_int, b=dores, out=any_int_or_res)
Register<1>(in=any_int_or_res, load=fetch, out=b_out) // Poll for interrupts at the end of the current instruction (during fetch)

// Interrupt vector address setting
// Forces lower bits of the ADL bus to 0 to jump to the correct vector
// FFFA (NMI), FFFC (RESET), FFFE (IRQ/BRK)
force_adl0 = brk5         // Active during T5 (low byte of vector)
force_adl1 = dores        // Active for RESET (FFFC)
force_adl2 = nmi_pending  // Active for NMI (FFFA)


// =====================================================================================================================
// DISPATCHER AND CYCLE COUNTERS
// =====================================================================================================================
Not<1>(in=rdy, out=n_rdy)

// An instruction is complete if it's a TWOCYCLE instruction currently in T1
And<1>(a=twocycle, b=t1, out=ends)  // Reset short instruction cycle counter

Or<1>(a=decoder_endx, b=brk7, out=endx)

Or<1>(a=ends, b=endx, out=fetch)

// Shift register for T0-T1
Mux<2>(a=state_t, b={t0, 1'b0}, sel=rdy, out=held_or_shifted_t)  // If ready, shift
Mux<2>(a=held_or_shifted_t, b=STATE_T0, sel=fetch, out=next_t)  // If fetch, force state back to T0
Dff<2>(in=next_t, out=state_t)
t0 = state_t[0]
t1 = state_t[1]

sync = t1

// Extended counter T2-T5: Triggers if T1 completes and it's not a 2-cycle instruction
And<1>(a=t1, b=n_twocycle, out=start_t2)
CycleCounter4(in_bit=start_t2, reset=fetch, hold=n_rdy, q0=t2, q1=t3, q2=t4, q3=t5)

// Interrupt sequence counter T6-T7 (BRK6E, BRK7)
// BRK takes 7 cycles. Trigger cycle 6 from T5 if an interrupt is active (b_out)
And<1>(a=b_out, b=t5, out=brk5)
Register<1>(in=brk5, load=rdy, out=brk6e)
Register<1>(in=brk6e, load=rdy, out=brk7)


// =====================================================================================================================
// INSTRUCTION REGISTER (IR)
// =====================================================================================================================

// Inject BRK (0x00) opcode for interrupt handling
Mux<8>(a=pd, b=BRK_OPCODE, sel=b_out, out=opcode)

And(a=fetch, b=phi1, out=ireg_load)
Register<8>(in=opcode, load=ireg_load, out=ireg) @IRegister


// =====================================================================================================================
// DECODER
// =====================================================================================================================
Decoder(
    ir=ireg
    n_rdy=n_rdy
    t0=t0, t1=t1, t2=t2, t3=t3, t4=t4, t5=t5
    dec_0_63=dec_0_63
    dec_64_127=dec_64_127
    dec_128_130=dec_128_130
)


// =====================================================================================================================
// RANDOM LOGIC
// =====================================================================================================================
RandomLogic(
    // Inputs
    dec_0_63=dec_0_63, dec_64_127=dec_64_127, dec_128_130=dec_128_130
    ireg=ireg
    t0=t0, t1=t1, t2=t2, t3=t3, t4=t4, t5=t5
    db=db_bus
    acr=acr, avr=avr
    nmi_pending=nmi_pending, irq_pending=irq_pending
    dores=dores, b_out=b_out, so=so, rdy=rdy

    // Feedback and flag outputs
    endx=decoder_endx
    i_flag=i_flag

    // Register hashes
    y_sb=hash_y_sb, x_sb=hash_x_sb, sb_y=hash_sb_y, sb_x=hash_sb_x
    s_sb=hash_s_sb, s_adl=hash_s_adl, sb_s=hash_sb_s

    // PC hashes
    adh_pch=hash_adh_pch, pch_adh=hash_pch_adh, pch_db=hash_pch_db
    adl_pcl=hash_adl_pcl, pcl_adl=hash_pcl_adl, pcl_db=hash_pcl_db
    inc_pc=hash_inc_pc

    // Bus and pin hashes
    adh_abh=hash_adh_abh, adl_abl=hash_adl_abl
    dl_adl=hash_dl_adl, dl_adh=hash_dl_adh, dl_db=hash_dl_db
    wr=hash_wr

    // ALU/AC hashes
    sb_ac=hash_sb_ac, ac_sb=hash_ac_sb, ac_db=hash_ac_db
    add_sb=hash_add_sb, add_adl=hash_add_adl
    ndb_add=hash_ndb_add, db_add=hash_db_add, zero_add=hash_0_add
    sb_add=hash_sb_add, adl_add=hash_adl_add
    acin=acin
    sums=hash_sums, ands=hash_ands, eors=hash_eors
    ors=hash_ors, srs=hash_srs
)


// =====================================================================================================================
// CONTROL COMMAND LATCHES
// =====================================================================================================================
Register<1>(in=hash_y_sb,  load=phi2, out=cmd_y_sb)  // Y => SB
Register<1>(in=hash_x_sb,  load=phi2, out=cmd_x_sb)  // X => SB
Register<1>(in=hash_sb_y,  load=phi2, out=cmd_sb_y)  // SB => Y
Register<1>(in=hash_sb_x,  load=phi2, out=cmd_sb_x)  // SB => X
Register<1>(in=hash_s_sb,  load=phi2, out=cmd_s_sb)  // S => SB
Register<1>(in=hash_s_adl, load=phi2, out=cmd_s_adl) // S => ADL
Register<1>(in=hash_sb_s,  load=phi2, out=cmd_sb_s)  // SB => S
Register<1>(in=hash_p_db, load=phi2, out=cmd_p_db) // P => DB

// The S/S command is active if the SB/S command is inactive
// This continuously refreshes the S register when it isn't being written to
Not<1>(in=cmd_sb_s, out=cmd_s_s)

Register<1>(in=hash_adh_pch, load=phi2, out=cmd_adh_pch) // ADH => PCH
Register<1>(in=hash_pch_adh, load=phi2, out=cmd_pch_adh) // PCH => ADH
Register<1>(in=hash_pch_db,  load=phi2, out=cmd_pch_db)  // PCH => DB
Register<1>(in=hash_adl_pcl, load=phi2, out=cmd_adl_pcl) // ADL => PCL
Register<1>(in=hash_pcl_adl, load=phi2, out=cmd_pcl_adl) // PCL => ADL
Register<1>(in=hash_pcl_db,  load=phi2, out=cmd_pcl_db)  // PCL => DB
Register<1>(in=hash_inc_pc,  load=phi2, out=cmd_inc_pc)  // 1 => PC (Abstracted as Active High)

// Address/Data bus latches
Register<1>(in=hash_adh_abh, load=phi2, out=cmd_adh_abh) // ADH => ABH
Register<1>(in=hash_adl_abl, load=phi2, out=cmd_adl_abl) // ADL => ABL
Register<1>(in=hash_dl_adl,  load=phi2, out=cmd_dl_adl)  // DL => ADL
Register<1>(in=hash_dl_adh,  load=phi2, out=cmd_dl_adh)  // DL => ADH
Register<1>(in=hash_dl_db,   load=phi2, out=cmd_dl_db)   // DL <=> DB
Register<1>(in=hash_wr,      load=phi2, out=cmd_wr)      // WR (Write Mode)

// ALU/AC latches
Register<1>(in=hash_sb_ac,   load=phi2, out=cmd_sb_ac)   // SB => AC
Register<1>(in=hash_ac_sb,   load=phi2, out=cmd_ac_sb)   // AC => SB
Register<1>(in=hash_ac_db,   load=phi2, out=cmd_ac_db)   // AC => DB
Register<1>(in=hash_add_sb,  load=phi2, out=cmd_add_sb)  // ADD => SB (Abstracts ADD/SB06 and ADD/SB7)
Register<1>(in=hash_add_adl, load=phi2, out=cmd_add_adl) // ADD => ADL

// ALU input and operation
Register<1>(in=hash_ndb_add, load=phi2, out=cmd_ndb_add)
Register<1>(in=hash_db_add,  load=phi2, out=cmd_db_add)
Register<1>(in=hash_0_add,   load=phi2, out=cmd_0_add)
Register<1>(in=hash_sb_add,  load=phi2, out=cmd_sb_add)
Register<1>(in=hash_adl_add, load=phi2, out=cmd_adl_add)
Register<1>(in=hash_sums,    load=phi2, out=cmd_sums)
Register<1>(in=hash_ands,    load=phi2, out=cmd_ands)
Register<1>(in=hash_eors,    load=phi2, out=cmd_eors)
Register<1>(in=hash_ors,     load=phi2, out=cmd_ors)
Register<1>(in=hash_srs,     load=phi2, out=cmd_srs)


// =====================================================================================================================
// PROGRAM COUNTER (PCL / PCH)
// =====================================================================================================================
Add<16>(a={pch_reg, pcl_reg}, b=16'd1, out=pc_inc)  // Incrementer

// PCL routing: Load from ADL, or load from Incrementer
Mux<8>(a=pcl_reg, b=pc_inc[0:7], sel=cmd_inc_pc, out=pcl_next_inc)
Mux<8>(a=pcl_next_inc, b=adl_bus, sel=cmd_adl_pcl, out=pcl_next)

// PCH routing: Load from ADH, or load from Incrementer
Mux<8>(a=pch_reg, b=pc_inc[8:15], sel=cmd_inc_pc, out=pch_next_inc)
Mux<8>(a=pch_next_inc, b=adh_bus, sel=cmd_adh_pch, out=pch_next)

// Load PCL/PCH during PHI1 if a command was active
Or<1>(a=cmd_inc_pc, b=cmd_adl_pcl, out=load_pcl)
And<1>(a=load_pcl, b=phi1, out=load_pcl_phi1)
Register<8>(in=pcl_next, load=load_pcl_phi1, out=pcl_reg) @PCL

Or<1>(a=cmd_inc_pc, b=cmd_adh_pch, out=load_pch)
And<1>(a=load_pch, b=phi1, out=load_pch_phi1)
Register<8>(in=pch_next, load=load_pch_phi1, out=pch_reg) @PCH


// =====================================================================================================================
// REGISTERS (X, Y, S): Loaded during PHI1
// =====================================================================================================================
// X Register
And<1>(a=cmd_sb_x, b=phi1, out=load_x)
Register<8>(in=sb_bus, load=load_x, out=x_reg) @XRegister

// Y Register
And<1>(a=cmd_sb_y, b=phi1, out=load_y)
Register<8>(in=sb_bus, load=load_y, out=y_reg) @YRegister

// S Register (Stack Pointer)
// The S register can output to both SB and ADL buses simultaneously
And<1>(a=cmd_sb_s, b=phi1, out=load_s)
Register<8>(in=sb_bus, load=load_s, out=s_reg) @SRegister


// =====================================================================================================================
// ARITHMETIC LOGIC UNIT (ALU)
// =====================================================================================================================
ALU(
    phi1=phi1, phi2=phi2
    sb_bus=sb_bus, db_bus=db_bus, adl_bus=adl_bus
    // Input commands
    ndb_add=cmd_ndb_add, db_add=cmd_db_add
    zero_add=cmd_0_add, sb_add=cmd_sb_add, adl_add=cmd_adl_add
    // Operations
    sums=cmd_sums, ands=cmd_ands, eors=cmd_eors
    ors=cmd_ors, srs=cmd_srs
    // Accumulator specific
    sb_ac=cmd_sb_ac
    acin=acin
    // Outputs
    ac_reg=ac_reg, add_reg=add_reg
    acr=acr, avr=avr
)


// =====================================================================================================================
// EXTERNAL PINS (ADDR, DATA, RW)
// =====================================================================================================================
// RW pin (1 = Read, 0 = Write)
Not<1>(in=cmd_wr, out=cmd_rd)
rw = cmd_rd

// Interrupts force lower bits of ADL to 0 before hitting ABL to jump to vectors
Not<1>(in=force_adl0, out=keep_adl0)
Not<1>(in=force_adl1, out=keep_adl1)
Not<1>(in=force_adl2, out=keep_adl2)

And<1>(a=adl_bus[0], b=keep_adl0, out=adl0_masked)
And<1>(a=adl_bus[1], b=keep_adl1, out=adl1_masked)
And<1>(a=adl_bus[2], b=keep_adl2, out=adl2_masked)

adl_masked = {adl_bus[7], adl_bus[6], adl_bus[5], adl_bus[4], adl_bus[3], adl2_masked, adl1_masked, adl0_masked}

// ABL (Address Bus Low) Register
And<1>(a=cmd_adl_abl, b=phi1, out=load_abl)
Register<8>(in=adl_masked, load=load_abl, out=abl_reg) @ABL

// ABH (Address Bus High) Register
And<1>(a=cmd_adh_abh, b=phi1, out=load_abh)
Register<8>(in=adh_bus, load=load_abh, out=abh_reg) @ABH

addr = {abh_reg, abl_reg}

// DL (Data Latch): Stores incoming data from external bus during PHI2
Register<8>(in=data_in, load=phi2, out=dl_reg) @DL

// DOR (Data Output Register): Stores data to be written during PHI1 if WR=1
And<1>(a=cmd_dl_db, b=cmd_wr, out=write_mode_dl)
And<1>(a=write_mode_dl, b=phi1, out=load_dor)
Register<8>(in=db_bus, load=load_dor, out=data_out) @DOR


// =====================================================================================================================

Neq<3>(a={dec_0_63[9], dec_0_63[8], dec_0_63[7]}, b=3'b0, out=x_sb_1)
Neq<5>(a={dec_0_63[14], dec_0_63[13], dec_0_63[12], dec_0_63[11], dec_0_63[10]}, b=5'b0, out=x_sb_2)

// SB => Y and SB => X
Neq<2>(a={dec_0_63[20], dec_0_63[19]}, b=2'b0, out=sb_y)
Neq<4>(a={dec_0_63[17], dec_0_63[16], dec_0_63[15], dec_0_63[14]}, b=4'b0, out=sb_x)

stk2 = dec_0_63[35] // Line 35

// PC CONTROL
// Controls moving the Program Counter to buses and incrementation
// =====================================================================================================================
// ADH => PCH

Not<1>(in=adh_pch, out=pch_adh) // PCH_ADH defaults when ADH_PCH is inactive

// ADL => PCL

Not<1>(in=adl_pcl, out=pcl_adl) // PCL_ADL defaults when ADL_PCL is inactive

// Increment PC (active mostly when fetch is occurring or advancing cycle)
// Note: Hardware uses active-low. Transpiler expects active-high representation here.

// =====================================================================================================================

// Governs ADL, ADH, and DB routing
// =====================================================================================================================

dl_adh = pch_db         // Reusing PCH_DB intermediate signal equivalent for DL_ADH

// Write command triggered primarily by STOR

// =====================================================================================================================
// ALU & ACCUMULATOR CONTROL

// =====================================================================================================================

Neq<2>(a={dec_64_127[4], dec_64_127[3]}, b=2'b0, out=ac_sb) // dec_67, 68
And<1>(a=dec_64_127[15], b=stor, out=ac_db)                 // STA (dec_79)

// ALU Input Multiplexing
Neq<3>(a={dec_0_63[51], dec_0_63[50], dec_0_63[49]}, b=3'b0, out=ndb_add)
Neq<3>(a={dec_0_63[54], dec_0_63[53], dec_0_63[52]}, b=3'b0, out=db_add)
Neq<3>(a={dec_64_127[2], dec_64_127[1], dec_64_127[0]}, b=3'b0, out=zero_add) // dec_64, 65, 66
Neq<3>(a={dec_64_127[8], dec_64_127[7], dec_64_127[6]}, b=3'b0, out=sb_add)   // dec_70, 71, 72
Neq<2>(a={dec_64_127[10], dec_64_127[9]}, b=2'b0, out=adl_add)                // dec_73, 74

Neq<2>(a={dec_0_63[14], dec_0_63[13]}, b=2'b0, out=add_sb)
Neq<2>(a={dec_64_127[18], dec_64_127[17]}, b=2'b0, out=add_adl)               // dec_81, 82

// ALU Operations
sums = db_add
Neq<2>(a={dec_64_127[6], dec_64_127[5]}, b=2'b0, out=ands) // dec_69, 70
eors = dec_0_63[29]
ors  = dec_0_63[32]
