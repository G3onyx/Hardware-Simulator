// ALU.hdl
CHIP ALU

IN
x[16], y[16]
zx // Zero the x input
nx // Negate the x input
zy // Zero the y input
ny // Negate the y input
f  // Function code: 1 for Add, 0 for And
no // Negate the out output

OUT
out[16]
zr // Zero flag:     True iff out=0
ng // Negative flag: True iff out<0

PARTS
Mux<16>(a=x, b=16'b0, sel=zx, out=xZeroed)
Not<16>(in=xZeroed, out=xNot)
Mux<16>(a=xZeroed, b=xNot, sel=nx, out=x1)

Mux<16>(a=y, b=16'b0, sel=zy, out=yZeroed)
Not<16>(in=yZeroed, out=yNot)
Mux<16>(a=yZeroed, b=yNot, sel=ny, out=y1)

Add<16>(a=x1, b=y1, out=xADDy)
And<16>(a=x1, b=y1, out=xANDy)
Mux<16>(a=xANDy, b=xADDy, sel=f, out=xy)

Not<16>(in=xy, out=nxy)
Mux<16>(a=xy, b=nxy, sel=no, out=output)

out = output
ng = output[15]

Eq<16>(a=output, b=16'd0, out=zr)

---

// CPU.hdl
CHIP CPU

IN 
instruction[16]  // Instruction to execute.
inM[16]          // Value of Mem[A], the instruction’s M input
reset            // Signals whether to continue (0) or restart (1)

OUT
outM[16]        // Value to write to Mem[addressM], the instruction’s M output
writeM          // Write to the Memory?
addressM[15]    // In which address to write?
pc[15]          // address of next instruction

PARTS

// Instruction decoding
isC = instruction[15]
Not<1>(in=isC, out=isA)


// Destination bits
And<1>(a=isC, b=instruction[5], out=destA)
Or<1>(a=isA, b=destA, out=loadA)

And<1>(a=isC, b=instruction[4], out=loadD)

And<1>(a=isC, b=instruction[3], out=writeM)


// Registers and routing

Mux<16>(a=aluOut, b=instruction, sel=isA, out=aIn)

Register(in=aIn, load=loadA, out=aOut) @ARegister

Register(in=aluOut, load=loadD, out=aluX) @DRegister
addressM = aOut[0:14]

Mux<16>(a=aOut, b=inM, sel=instruction[12], out=aluY)


// ALU
ALU(
    x=aluX, y=aluY,
    zx=instruction[11], nx=instruction[10]
    zy=instruction[9], ny=instruction[8]
    f=instruction[7], no=instruction[6]
    out=aluOut
    zr=zr, ng=ng
)
outM = aluOut


// Jump logic

Or<1>(a=ng, b=zr, out=ngOrZr)
Not<1>(in=ngOrZr, out=pos)

And<1>(a=instruction[2], b=ng, out=jlt) // JLT
And<1>(a=instruction[1], b=zr, out=jeq) // JEQ
And<1>(a=instruction[0], b=pos, out=jgt) // JGT

Or<1>(a=jlt, b=jeq, out=jle)
Or<1>(a=jle, b=jgt, out=jumpCond)
And<1>(a=isC, b=jumpCond, out=jump)


// Program counter
PC(in=aOut, inc=1'b1, load=jump, reset=reset, out=pcOut) @PC
pc = pcOut[0:14]
---

CHIP HackComputer

@toplevel

IN reset

PARTS

CPU(
    instruction=instruction
    inM=memoryOut
    reset=reset
    outM=outM
    writeM=writeM
    addressM=addressM
    pc=pc
)

// RAM + Screen + Keyboard
Memory(in=outM, address=addressM, load=writeM, out=memoryOut)

ROM<15, 16>(address=pc, out=instruction) @ROM32K
---
